    # .profile

    function get_branch {
        branch=$(git branch 2>/dev/null | grep '^*' | tr -d ' *')
        if [ -n "${branch}" ]; then
            echo "[${branch}]"
        fi
    }

    export PS1="\u@\h:\w\$(get_branch)\$ "

    alias tmux='tmux -2'

    export EDITOR=vim
    export PAGER=less

    export CXXFLAGS="-std=c++26 -Wall -Werror -Wfatal-errors"

    export BOARD=arduino_uno_r3
    export PORT=/dev/ttyACM0
    export BAUD=115200
