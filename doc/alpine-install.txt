    jfathman@alpine:~/work/uno$ cat blink.cpp 
    // blink.cpp

    #ifndef F_CPU
    #define F_CPU 16000000UL
    #endif

    #include <avr/io.h>
    #include <util/delay.h>

    int main(void)
    {
            DDRB |= (1 << PB5);       // Pin 13 (PB5) output

            while (1) {
                    PORTB |= (1 << PB5);  // LED On
                    _delay_ms(500);

                    PORTB &= ~(1 << PB5); // LED Off
                    _delay_ms(500);
            }

            return 0;
    }
